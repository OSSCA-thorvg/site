마스크와 Scene을 각각 렌더한 뒤 마스크 알파로 합성합니다.

```cpp
// Canvas target: transparent 16 × 12 ABGR8888 pixels.
// photoPixels: decoded opaque 16 × 12 ABGR8888 pixels.
auto avatar = Scene::gen();
auto photo = Picture::gen();
photo->load(photoPixels, 16, 12, ColorSpace::ABGR8888, true);
auto badge = Shape::gen();
badge->appendCircle(12, 8.5, 2, 2);
badge->fill(35, 160, 100);
avatar->add(photo);
avatar->add(badge);

auto fadeMask = Shape::gen();
fadeMask->appendCircle(8, 5.5, 5, 5);
fadeMask->fill(45, 160, 210, 160);
avatar->mask(fadeMask, MaskMethod::Alpha);
canvas->add(avatar);
canvas->update();
canvas->draw(true);
canvas->sync();
```
